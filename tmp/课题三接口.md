 `GET /api/download/{filename}` — 下载压缩包
获取文件示例，

import os
import requests

BASE = "http://10.135.40.250:50053"

# 第一步：打包
r = requests.post(f"{BASE}/api/verify",
                  json={"project_id": "1001"},   # 或 {"dist_dir": "..."}
                  timeout=600)                    # 同步阻塞，超时给足
r.raise_for_status()                              # 4xx/404 会在这里抛
info = r.json()
assert info.get("success"), info

zip_url = info["zip_url"]                         # /api/download/NWM_synthesis.zip
fname   = os.path.basename(zip_url)               # NWM_synthesis.zip

# 第二步：流式下载
with requests.get(f"{BASE}{zip_url}", stream=True, timeout=600) as resp:
    resp.raise_for_status()
    with open(fname, "wb") as f:                  # 必须 "wb"，zip 是二进制
        for chunk in resp.iter_content(chunk_size=8192):
            f.write(chunk)

print(f"已下载 {fname}  {os.path.getsize(fname)} 字节")
