\section{AG Contract}

\begin{description}
\item[Assumptions]:
\begin{align*}
Assumptions \definedby \
& \syntax{len}(\Var{StateFlag}) \geq 9
\end{align*}

\item[Guarantees]:
\begin{align*}
Guarantees \definedby \
& Check_1 \wedge Check_2 \wedge Check_3 \\
Check_1 \definedby \
& \syntax{len}(\Var{SignFlag}') = 9 \\
& \wedge \clock'=\clock \\
Check_2 \definedby \
& \forall \Var{JoinTotal}' \leq i < 9 \cdot \Var{SignFlag}'[i] = 0 \\
Check_3 \definedby \
& \exists \Var{tempFlag} : \mathbb{N} \rightharpoonup \mathbb{R} \cdot
\big{(} \
\syntax{len}(\Var{tempFlag}) = 9 \wedge Check_4 \wedge Check_5 \wedge Check_6 \wedge Check_7 \
\big{)} \\
Check_4 \definedby \
& \forall 0 \leq i < 9 \cdot \big{(} \ \Var{tempFlag}[i] = 1 \lhd \Var{StateFlag}[i] = \texttt{0xEB} \rhd \Var{tempFlag}[i] = 0 \ \big{)} \\
Check_5 \definedby \
& \Var{JoinTotal}' = \textstyle \sum_{j=0}^8 \Var{tempFlag}[j] \\
Check_6 \definedby \
& \forall 0 \leq i < \Var{JoinTotal}' \cdot \Var{SignFlag}'[i] = \min \{ k \mid \textstyle \sum_{j=0}^k \Var{tempFlag}[j] = i \} \\
Check_7 \definedby \
& \Var{gyroStatus0}' = \textstyle \sum_{i=0}^8 \Var{tempFlag}[i] \times 2^i
\end{align*}

\end{description}
