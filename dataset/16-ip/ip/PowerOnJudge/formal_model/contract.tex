\section{AG Contract}

\begin{description}

\item[Assumptions]:
\begin{align*}
Assumptions \definedby \
& \Var{powerState}=0 \\
& \vee \Var{powerState} = 255
\end{align*}

\item[Guarantees]:
\begin{align*}
Guarantees \definedby \
& (Case_1 \vee Case_2 \vee Case_3 \vee Case_4 \vee Case_5) \wedge Unchange \\
Case_1 \definedby \
& \Var{powerState} = 0 \\
& \wedge \Var{cntPowerOff} < \Var{numOffPeriod} \\
& \wedge \Var{powerPr1}' = 0 \\
& \wedge \Var{cntPowerOn}' = \Var{cntPowerOn} \\
& \wedge \Var{cntPowerOff}' = \Var{cntPowerOff} + 1 \\
Case_2 \definedby \
& \Var{powerState} = 0 \\
& \wedge \Var{cntPowerOff} \geq \Var{numOffPeriod} \\
& \wedge \neg \Var{bPowerOnOk}' \\
& \wedge \Var{powerPr1}' = 0 \\
& \wedge \Var{cntPowerOn}' = 0 \\
& \wedge \Var{cntPowerOff}' = \Var{cntPowerOff} \\
Case_3 \definedby \
& \Var{powerPr1} = 0 \\
& \wedge \Var{powerState} = 255 \\
& \wedge \Var{cntPowerOn}' > \Var{numOnPeriod} \\
& \wedge \Var{bPowerOnOk}' \\
& \wedge \Var{powerPr1}' = 255 \\
& \wedge \Var{cntPowerOn}' = \Var{cntPowerOn} + 1 \\
& \wedge \Var{cntPowerOff}' = 0 \\
Case_4 \definedby \
& \Var{powerPr1}=0 \\
& \wedge \Var{powerState} = 255 \\
& \wedge \Var{cntPowerOn}' \leq \Var{numOnPeriod} \\
& \wedge \Var{powerPr1}' = 0 \\
& \wedge \Var{cntPowerOn}' = \Var{cntPowerOn} + 1 \\
& \wedge \Var{cntPowerOff}' = \Var{cntPowerOff} \\
Case_5 \definedby \
& \Var{powerState}\neq 0 \\
& \wedge (\Var{powerPr1} \neq 0 \vee \Var{powerState} \neq 255) \\
& \wedge \Var{powerPr1}' = \Var{powerState} \\
& \wedge \Var{cntPowerOn}' = \Var{cntPowerOn} \\
& \wedge \Var{cntPowerOff}' = \Var{cntPowerOff} \\
Unchange \definedby \
& \Var{numOnPeriod}' = \Var{numOnPeriod} \\
& \wedge \Var{numOffPeriod}' = \Var{numOffPeriod}
\end{align*}

\end{description}
