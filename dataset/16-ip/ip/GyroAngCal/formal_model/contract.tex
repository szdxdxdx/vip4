\section{AG Contract}

\begin{description}
\item[Assumptions]:
\begin{align*}
Assumptions \definedby \
& \syntax{len}(\Var{W}) \geq \syntax{len}(\Var{Gi}) = 3
\end{align*}

\item[Guarantees]:
\begin{align*}
Guarantees \definedby \
& \syntax{len}(\Var{Gi\_out}') = \syntax{len}(\Var{Gi}') = 3 \\
& \wedge \Var{Gi\_out}'[0] = \Var{Gi}'[0] = \max\{-180, \min\{\Var{Gi}[0] + \Var{W}[0] \times \Var{ctlPeriod}, 180\}\} \\
& \wedge \Var{Gi\_out}'[1] = \Var{Gi}'[1] = \max\{-180, \min\{\Var{Gi}[1] + \Var{W}[1] \times \Var{ctlPeriod}, 180\}\} \\
& \wedge \Var{Gi\_out}'[2] = \Var{Gi}'[2] = \max\{-90,  \min\{\Var{Gi}[2] + \Var{W}[2] \times \Var{ctlPeriod}, 90\} \} \\
& \wedge \Var{ctlPeriod}' = \Var{ctlPeriod} \\
& \wedge \clock' = \clock
\end{align*}

\end{description}
