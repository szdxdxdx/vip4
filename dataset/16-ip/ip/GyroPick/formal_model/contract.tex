\section{AG Contract}

\begin{description}
\item[Assumptions]:
\begin{align*}
Assumptions \definedby \
& \syntax{len}(\Var{wa}) \geq 9 \\
& \wedge \syntax{len}(\Var{wal}) = \syntax{len}(\Var{countPick}) = 9
\end{align*}

\item[Guarantees]:
\begin{align*}
Guarantees \definedby \
& Check_1 \wedge Check_2 \\
Check_1 \definedby \
& \syntax{len}(\Var{wa\_out}') = \syntax{len}(\Var{wal}') = \syntax{len}(\Var{countPick}') = 9 \\
& \wedge \Var{waThr}' = \Var{waThr} \\
& \wedge \Var{pickThr}' = \Var{pickThr} \\
& \wedge \clock'=\clock \\
Check_2 \definedby \
& \forall 0 \leq i < 9 \cdot
\left(
\begin{aligned}
& \syntax{if } |\Var{wa}[i]-\Var{wal}[i]| > \Var{waThr} \ \wedge \ \Var{countPick}[i]+1 < \Var{pickThr} \syntax{ then} \\
& \quad \Var{wa\_out}'[i] = \Var{wal}'[i] = \Var{wal}[i] \\
& \quad \wedge \Var{countPick}'[i] = \Var{countPick}[i]+1 \\
& \syntax{else} \\
& \quad \Var{wa\_out}'[i] = \Var{wal}'[i] = \Var{wa}[i] \\
& \quad \wedge \Var{countPick}'[i] = 0 \\
& \syntax{endif}
\end{aligned}
\right)
\end{align*}

\end{description}
