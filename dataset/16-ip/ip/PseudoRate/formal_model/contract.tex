\section{AG Contract}

\begin{description}
\item[Assumptions]:
\begin{align*}
Assumptions \definedby \
& \syntax{len}(\Var{pu}) \geq 3 \\
& \wedge \syntax{len}(\Var{r}) = \syntax{len}(\Var{h1}) = 3
\end{align*}
\item[Guarantees]:
\begin{align*}
Guarantees \definedby \
& Check_1 \wedge Check_2 \\
Check_1 \definedby \
& \syntax{len}(\Var{Yp}') = \syntax{len}(\Var{Yn}') = \syntax{len}(\Var{r}') = \syntax{len}(\Var{h1}') = 3 \\
& \wedge \Var{h1}'=\Var{h1} \\
& \wedge \clock'=\clock \\
Check_2 \definedby \
& \forall 0 \leq i < 3 \cdot 
\left(
\begin{aligned}
& \syntax{if } \Var{pu}[i] - \Var{r}[i] > \Var{h1}[i] \syntax{ then} \\
& \quad \Var{Yp}'[i] = \texttt{0x0} \\
& \quad\wedge \Var{Yn}'[i] = \texttt{0xF} \\
& \quad\wedge \Var{r}'[i] = 0.9231 \times \Var{r}[i] + 0.07688 \\
& \syntax{elif } \Var{pu}[i] - \Var{r}[i] < - \Var{h1}[i] \syntax{ then} \\
& \quad \Var{Yp}'[i] = \texttt{0xF} \\
& \quad \wedge \Var{Yn}'[i] = \texttt{0x0} \\
& \quad \wedge \Var{r}'[i] = 0.9231 \times \Var{r}[i] - 0.07688 \\
& \syntax{else} \\
& \quad \Var{Yp}'[i] = \texttt{0x0} \\
& \quad \wedge \Var{Yn}'[i] = \texttt{0x0} \\
& \quad \wedge \Var{r}'[i] = 0.9231 \times \Var{r}[i] \\
& \syntax{endif}
\end{aligned}
\right)
\end{align*}
\end{description}
